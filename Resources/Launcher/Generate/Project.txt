cmake_minimum_required(VERSION 3.6 FATAL_ERROR)

project("PLACEHOLDER_PROJECT_NAME")

set(CMAKE_CXX_STANDARD 23)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(CMAKE_DEBUG_POSTFIX "")

#Common
set(BLOODSHOT_MAIN_DIR "PLACEHOLDER_BLOODSHOT_DIR")
set(BLOODSHOT_BINARIES_DIR ${BLOODSHOT_MAIN_DIR}/Binaries)
set(BLOODSHOT_INTERMEDIATE_DIR ${BLOODSHOT_MAIN_DIR}/Intermediate)
set(BLOODSHOT_SOURCE_DIR ${BLOODSHOT_MAIN_DIR}/Source)
set(BLOODSHOT_THIRD_PARTY_DIR ${BLOODSHOT_MAIN_DIR}/ThirdParty)
set(BLOODSHOT_PROGRAMS_DIR ${BLOODSHOT_SOURCE_DIR}/Programs)
set(BLOODSHOT_SHARED_DIR ${BLOODSHOT_SOURCE_DIR}/Shared)
set(BLOODSHOT_CMAKE_MODULE_DIR ${BLOODSHOT_MAIN_DIR}/Shared/cmake)
set(${PROJECT_NAME}_PROJECT_FOLDER_NAME ${PROJECT_NAME})
set(${PROJECT_NAME}_BINARIES_DIR ${CMAKE_SOURCE_DIR}/Binaries)
set(${PROJECT_NAME}_INTERMEDIATE_DIR ${CMAKE_SOURCE_DIR}/Intermediate)
set(${PROJECT_NAME}_SOURCE_DIR ${CMAKE_SOURCE_DIR}/Source)

#CMake Modules
include(${BLOODSHOT_CMAKE_MODULE_DIR}/Macros.cmake)

execute_process(COMMAND ${CMAKE_COMMAND} -E chdir ${CMAKE_SOURCE_DIR}
	"${BLOODSHOT_BINARIES_DIR}/Bloodshot Header Tool.exe" 
	Source:${${PROJECT_NAME}_SOURCE_DIR}
	Output:${${PROJECT_NAME}_INTERMEDIATE_DIR}
	COMMENT "Run Bloodshot Header Tool on ${PROJECT_NAME} sources")

#Modules
file(GLOB USER_MODULE_DIRS ${${PROJECT_NAME}_SOURCE_DIR}/*)
foreach(USER_MODULE_DIR ${USER_MODULE_DIRS})
	add_subdirectory(${USER_MODULE_DIR})
endforeach()
